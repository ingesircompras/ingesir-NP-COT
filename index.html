function generarDocumento() { const proveedor = document.getElementById('inputProveedor').value; const tipoDoc = document.getElementById('selectTipo').value; const filas = document.querySelectorAll('#cuerpoTabla tr'); let itemsParaGuardar = []; filas.forEach(fila => { itemsParaGuardar.push({ desc: fila.querySelector('.desc-item').innerText, cant: fila.querySelector('.cant-item').value }); }); if (proveedor === '' || itemsParaGuardar.length === 0) { alert('Falta ingresar el nombre del proveedor o agregar artículos.'); return; } // Guardamos en memoria local de tu navegador por si querés recuperar la lista después localStorage.setItem('ingesir_borrador', JSON.stringify(itemsParaGuardar)); // Armamos la caja con los datos para mandarla por internet const datos = { proveedor: proveedor, tipoDoc: tipoDoc, items: itemsParaGuardar }; // Efecto visual: Cambiamos el texto del botón para que sepas que está procesando const btn = document.querySelector('.btn-success'); btn.innerHTML = "⏳ GENERANDO PDF EN EL DRIVE..."; btn.disabled = true; // --- REEMPLAZÁ ESTO POR LA URL QUE COPIASTE DE GOOGLE --- const urlGoogle = 'PEGA_TU_URL_AQUI_ADENTRO'; // Enviamos los datos fetch(urlGoogle, { method: 'POST', mode: 'no-cors', // Evita bloqueos de seguridad del navegador headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(datos) }) .then(() => { alert("✅ ¡Éxito!\n\nLos datos se enviaron a tu Google Sheets y el PDF se guardó en tu carpeta de Drive."); btn.innerHTML = "📄 GENERAR DOCUMENTO"; btn.disabled = false; }) .catch(error => { alert("Hubo un problema de conexión. Revisá tu internet o el enlace de Google."); btn.innerHTML = "📄 GENERAR DOCUMENTO"; btn.disabled = false; }); }
