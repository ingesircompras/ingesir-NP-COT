function generarDocumento() { const proveedor = document.getElementById('inputProveedor').value; const tipoDoc = document.getElementById('selectTipo').value; const filas = document.querySelectorAll('#cuerpoTabla tr'); let itemsParaGuardar = []; filas.forEach(fila => { itemsParaGuardar.push({ desc: fila.querySelector('.desc-item').innerText, cant: fila.querySelector('.cant-item').value }); }); if (proveedor === '' || itemsParaGuardar.length === 0) { alert('Falta ingresar el nombre del proveedor o agregar artículos.'); return; } guardarEnHistorial(proveedor, tipoDoc, itemsParaGuardar); const datos = { proveedor: proveedor, tipoDoc: tipoDoc, items: itemsParaGuardar }; const btn = document.querySelector('.btn-success'); btn.innerHTML = "⏳ GENERANDO PDF EN EL DRIVE..."; btn.disabled = true; const urlGoogle = 'https://script.google.com/macros/s/AKfycbwarkcDCdb4dTlC9UtUi0OS4bHRQWvo2BvO3kEeX4klKuph1H4WhiNI_CMd1j8ZnI9V/exec'; fetch(urlGoogle, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(datos) }) .then(response => response.text()) .then(resultado => { alert("Respuesta del servidor:\n" + resultado); btn.innerHTML = "📄 GENERAR DOCUMENTO"; btn.disabled = false; // Limpieza automática document.getElementById('inputProveedor').value = ''; document.getElementById('cuerpoTabla').innerHTML = ''; document.getElementById('inputDesc').value = ''; document.getElementById('inputCant').value = ''; document.getElementById('inputProveedor').focus(); }) .catch(error => { alert("Error de red: " + error); btn.innerHTML = "📄 GENERAR DOCUMENTO"; btn.disabled = false; }); }
